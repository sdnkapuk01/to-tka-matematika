                explanation: "Total tabungan Dira = 45.000 + 55.000 + 50.000 + 70.000 = Rp220.000. Total tabungan Farel = 4 × Rp52.500 = Rp210.000. Jadi, total tabungan Dira lebih banyak Rp10.000 (220.000 - 210.000) daripada Farel."
            }
        ];

        let userAnswers = new Array(questionsData.length).fill(null);

        // =========================================================
        // UTILITAS UI (TOAST & MODAL KONFIRMASI)
        // =========================================================
        function showToast(message, type = 'info') {
            const container = document.getElementById('toast-container');
            const toast = document.createElement('div');
            
            const colors = {
                error: 'bg-rose-600 text-white',
                success: 'bg-emerald-600 text-white',
                info: 'bg-slate-800 text-white'
            };
            const icons = {
                error: 'fa-circle-exclamation',
                success: 'fa-circle-check',
                info: 'fa-circle-info'
            };

            toast.className = `${colors[type] || colors.info} px-4 py-3 rounded-xl shadow-lg flex items-center gap-3 text-sm font-medium transform transition-all duration-300 translate-y-0 opacity-100 pointer-events-auto`;
            toast.innerHTML = `<i class="fa-solid ${icons[type] || icons.info}"></i><span>${message}</span>`;
            
            container.appendChild(toast);
            setTimeout(() => {
                toast.classList.add('opacity-0', '-translate-y-2');
                setTimeout(() => toast.remove(), 300);
            }, 3200);
        }

        function showConfirmModal(title, message, onConfirm) {
            const modal = document.getElementById('confirm-modal');
            document.getElementById('confirm-title').textContent = title;
            document.getElementById('confirm-message').textContent = message;
            
            modal.classList.remove('hidden');
            modal.classList.add('flex');

            const btnCancel = document.getElementById('confirm-cancel');
            const btnOk = document.getElementById('confirm-ok');

            const cleanup = () => {
                modal.classList.add('hidden');
                modal.classList.remove('flex');
                btnCancel.onclick = null;
                btnOk.onclick = null;
            };

            btnCancel.onclick = cleanup;
            btnOk.onclick = () => {
                cleanup();
                onConfirm();
            };
        }

        document.getElementById('input-nisn').addEventListener('input', function() {
            this.value = this.value.replace(/[^0-9]/g, '');

        document.getElementById('student-form').addEventListener('submit', function(e) {
            e.preventDefault();
            const nisn = document.getElementById('input-nisn').value.trim();
            const nama = document.getElementById('input-nama').value.trim();
            const kelas = document.getElementById('input-kelas').value;

            if (nisn.length < 5) {
                showToast('Mohon masukkan nomor NISN yang valid (minimal 5-10 digit angka).', 'error');
                return;
            }
            if (!nama || !kelas) {
                showToast('Mohon lengkapi Nama dan Kelas terlebih dahulu.', 'error');
                return;
            }

            studentData = { nisn, nama, kelas };
            
            document.getElementById('display-nama').textContent = studentData.nama;
            document.getElementById('display-nisn').textContent = studentData.nisn;
            document.getElementById('display-kelas').textContent = studentData.kelas;
            document.getElementById('avatar-initials').textContent = studentData.nama.charAt(0).toUpperCase();

            document.getElementById('view-login').classList.add('hidden');
            document.getElementById('view-quiz').classList.remove('hidden');
            window.scrollTo({ top: 0, behavior: 'smooth' });

            initQuiz();
            showToast(`Selamat mengerjakan 30 soal, ${studentData.nama}!`, 'success');
        });

        // =========================================================
        // LOGIKA HALAMAN 2: PENGERJAAN SOAL
        // =========================================================
        function initQuiz() {
            currentQuestionIndex = 0;
            userAnswers = new Array(questionsData.length).fill(null);
            renderNavGrid();
            renderQuestion(0);
            updateProgress();
        }

        function renderNavGrid() {
            const grid = document.getElementById('question-nav-grid');
            grid.innerHTML = '';

            questionsData.forEach((q, idx) => {
                const btn = document.createElement('button');
                btn.type = 'button';
                btn.textContent = idx + 1;
                btn.id = `nav-btn-${idx}`;
                btn.onclick = () => renderQuestion(idx);
                
                let baseClass = "h-9 rounded-lg font-display font-bold text-xs transition flex items-center justify-center border ";
                if (idx === currentQuestionIndex) {
                    baseClass += "bg-primary-600 text-white border-primary-600 shadow-md shadow-primary-600/25 ring-2 ring-primary-200";
                } else if (userAnswers[idx] !== null) {
                    baseClass += "bg-emerald-50 text-emerald-700 border-emerald-300 hover:bg-emerald-100";
                } else {
                    baseClass += "bg-slate-50 text-slate-600 border-slate-200 hover:bg-slate-100";
                }
                
                btn.className = baseClass;
                grid.appendChild(btn);
            });
        }

        function renderQuestion(index) {
            currentQuestionIndex = index;
            const q = questionsData[index];

            document.getElementById('q-number').textContent = q.id;
            document.getElementById('q-domain').textContent = `${q.domain} • ${q.subDomain}`;
            document.getElementById('q-level').textContent = `Level: ${q.level}`;
            document.getElementById('q-text').innerHTML = q.question;

            // Render Ilustrasi Grafis jika ada
            const graphicContainer = document.getElementById('q-graphic-container');
            const graphicBox = document.getElementById('q-graphic');
            const graphicCaption = document.getElementById('q-graphic-caption');

            if (q.graphicSVG) {
                graphicContainer.classList.remove('hidden');
                graphicBox.innerHTML = q.graphicSVG;
                graphicCaption.textContent = q.graphicCaption || '';
            } else {
                graphicContainer.classList.add('hidden');
                graphicBox.innerHTML = '';
            }

            // Render Opsi Pilihan Ganda
            const optionsContainer = document.getElementById('q-options');
            optionsContainer.innerHTML = '';
            const letters = ['A', 'B', 'C', 'D'];

            q.options.forEach((optText, optIdx) => {
                const isSelected = userAnswers[index] === optIdx;
                const card = document.createElement('button');
                card.type = 'button';
                card.onclick = () => selectOption(index, optIdx);

                card.className = `option-card w-full text-left p-4 rounded-2xl border-2 flex items-start gap-3.5 transition ${
                    isSelected 
                    ? 'border-primary-600 bg-primary-50/70 shadow-sm' 
                    : 'border-slate-200 bg-white hover:border-primary-300 hover:bg-slate-50'
                }`;

                card.innerHTML = `
                    <span class="w-8 h-8 rounded-lg font-display font-bold text-sm flex items-center justify-center shrink-0 transition ${
                        isSelected 
                        ? 'bg-primary-600 text-white' 
                        : 'bg-slate-100 text-slate-600'
                    }">${letters[optIdx]}</span>
                    <span class="text-sm sm:text-base text-slate-800 font-medium pt-1 leading-snug">${optText}</span>
                `;

                optionsContainer.appendChild(card);
            });

            // Atur tombol Prev / Next / Finish
            document.getElementById('btn-prev').disabled = (index === 0);
            
            const btnNext = document.getElementById('btn-next');
            const btnFinishInline = document.getElementById('btn-finish-inline');

            if (index === questionsData.length - 1) {
                btnNext.classList.add('hidden');
                btnFinishInline.classList.remove('hidden');
                btnFinishInline.classList.add('flex');
            } else {
                btnNext.classList.remove('hidden');
                btnFinishInline.classList.add('hidden');
                btnFinishInline.classList.remove('flex');
            }

            renderNavGrid();
        }

        function selectOption(qIndex, optIndex) {
            userAnswers[qIndex] = optIndex;
            renderQuestion(qIndex);
            updateProgress();
        }

        function updateProgress() {
            const answered = userAnswers.filter(a => a !== null).length;
            const total = questionsData.length;
            const pct = Math.round((answered / total) * 100);

            document.getElementById('answered-count').textContent = answered;
            document.getElementById('progress-bar').style.width = `${pct}%`;
        }

        document.getElementById('btn-prev').addEventListener('click', () => {
            if (currentQuestionIndex > 0) renderQuestion(currentQuestionIndex - 1);
        });

        document.getElementById('btn-next').addEventListener('click', () => {
            if (currentQuestionIndex < questionsData.length - 1) renderQuestion(currentQuestionIndex + 1);
        });

        function handleFinishAttempt() {
            const unansweredCount = userAnswers.filter(a => a === null).length;
            const msg = unansweredCount > 0 
                ? `Masih ada ${unansweredCount} dari 30 soal yang belum kamu jawab. Apakah kamu yakin ingin menyelesaikan Try Out sekarang?`
                : `Semua 30 soal sudah dijawab! Apakah kamu yakin ingin mengumpulkan jawaban dan melihat laporan hasil evaluasi?`;

            showConfirmModal('Selesaikan Try Out?', msg, () => {
                generateEvaluationReport();
            });
        }

        document.getElementById('btn-finish-inline').addEventListener('click', handleFinishAttempt);
        document.getElementById('btn-submit-sidebar').addEventListener('click', handleFinishAttempt);

        // =========================================================
        // LOGIKA HALAMAN 3: EVALUASI & EXPORT PDF
        // =========================================================
        function generateEvaluationReport() {
            document.getElementById('view-quiz').classList.add('hidden');
            document.getElementById('view-result').classList.remove('hidden');
            window.scrollTo({ top: 0, behavior: 'smooth' });

            const now = new Date();
            const optionsDate = { weekday: 'long', year: 'numeric', month: 'long', day: 'numeric' };
            document.getElementById('report-date').textContent = now.toLocaleDateString('id-ID', optionsDate);

            document.getElementById('res-nama').textContent = `: ${studentData.nama}`;
            document.getElementById('res-nisn').textContent = `: ${studentData.nisn}`;
            document.getElementById('res-kelas').textContent = `: ${studentData.kelas} (SDN Kapuk 01)`;

            let correctCount = 0;
            let wrongCount = 0;
            let emptyCount = 0;

            const domainStats = {
                "Bilangan": { correct: 0, total: 0 },
                "Geometri dan Pengukuran": { correct: 0, total: 0 },
                "Pengolahan Data": { correct: 0, total: 0 }
            };

            questionsData.forEach((q, idx) => {
                const uAns = userAnswers[idx];
                if (domainStats[q.domain]) {
                    domainStats[q.domain].total++;
                }

                if (uAns === null) {
                    emptyCount++;
                } else if (uAns === q.answer) {
                    correctCount++;
                    if (domainStats[q.domain]) {
                        domainStats[q.domain].correct++;
                    }
                } else {
                    wrongCount++;
                }
            });

            // Skor Skala 100 (dibulatkan 1 desimal jika perlu atau bilangan bulat)
            const rawScore = (correctCount / questionsData.length) * 100;
            const finalScore = Number.isInteger(rawScore) ? rawScore : rawScore.toFixed(1);
            document.getElementById('res-score').textContent = finalScore;
            document.getElementById('res-correct').textContent = correctCount;
            document.getElementById('res-wrong').textContent = wrongCount;
            document.getElementById('res-empty').textContent = emptyCount;

            const predikatEl = document.getElementById('res-predikat');
            if (rawScore >= 90) {
                predikatEl.innerHTML = `: <span class="text-emerald-600">Sangat Baik (A) — Siap TKA!</span>`;
            } else if (rawScore >= 75) {
                predikatEl.innerHTML = `: <span class="text-primary-600">Baik (B) — Pertahankan!</span>`;
            } else if (rawScore >= 60) {
                predikatEl.innerHTML = `: <span class="text-amber-600">Cukup (C) — Perlu Latihan Tambahan</span>`;
            } else {
                predikatEl.innerHTML = `: <span class="text-rose-600">Perlu Bimbingan Intensif (D)</span>`;
            }

            // Render Analisis Domain
            const domainContainer = document.getElementById('domain-analysis');
            domainContainer.innerHTML = '';
            Object.keys(domainStats).forEach(domName => {
                const stat = domainStats[domName];
                const pct = Math.round((stat.correct / stat.total) * 100);
                let barColor = pct >= 75 ? 'bg-emerald-500' : (pct >= 50 ? 'bg-amber-500' : 'bg-rose-500');

                domainContainer.innerHTML += `
                    <div class="p-4 rounded-2xl border border-slate-200 bg-white">
                        <div class="text-xs font-bold text-slate-700 mb-1">${domName}</div>
                        <div class="flex items-end justify-between mb-2">
                            <span class="font-display font-extrabold text-xl text-slate-900">${pct}%</span>
                            <span class="text-xs text-slate-500">Benar ${stat.correct} dari ${stat.total} soal</span>
                        </div>
                        <div class="w-full h-2 bg-slate-100 rounded-full overflow-hidden">
                            <div class="${barColor} h-full" style="width: ${pct}%"></div>
                        </div>
                    </div>
                `;
            });

            // Render Rincian Pembahasan 30 Soal
            const reviewList = document.getElementById('review-list');
            reviewList.innerHTML = '';
            const letters = ['A', 'B', 'C', 'D'];

            questionsData.forEach((q, idx) => {
                const uAns = userAnswers[idx];
                const isCorrect = (uAns === q.answer);
                const isEmpty = (uAns === null);

                let statusBadge = '';
                let cardBorder = '';
                if (isCorrect) {
                    statusBadge = `<span class="px-2.5 py-0.5 rounded-full text-xs font-bold bg-emerald-100 text-emerald-700"><i class="fa-solid fa-check mr-1"></i>Benar</span>`;
                    cardBorder = 'border-emerald-200 bg-emerald-50/20';
                } else if (isEmpty) {
                    statusBadge = `<span class="px-2.5 py-0.5 rounded-full text-xs font-bold bg-slate-200 text-slate-700"><i class="fa-solid fa-minus mr-1"></i>Tidak Dijawab</span>`;
                    cardBorder = 'border-slate-200 bg-slate-50/40';
                } else {
                    statusBadge = `<span class="px-2.5 py-0.5 rounded-full text-xs font-bold bg-rose-100 text-rose-700"><i class="fa-solid fa-xmark mr-1"></i>Salah</span>`;
                    cardBorder = 'border-rose-200 bg-rose-50/20';
                }

                const studentAnswerText = isEmpty 
                    ? '<span class="italic text-slate-400">Tidak dijawab</span>' 
                    : `<strong>${letters[uAns]}.</strong> ${q.options[uAns]}`;
                const keyAnswerText = `<strong>${letters[q.answer]}.</strong> ${q.options[q.answer]}`;

                reviewList.innerHTML += `
                    <div class="p-4 sm:p-5 rounded-2xl border ${cardBorder} text-sm space-y-2.5 break-inside-avoid">
                        <div class="flex flex-wrap items-center justify-between gap-2">
                            <div class="flex items-center gap-2">
                                <span class="font-display font-bold text-slate-900">Soal No. ${q.id}</span>
                                <span class="text-xs px-2 py-0.5 rounded bg-slate-100 text-slate-600">${q.domain} (${q.subDomain})</span>
                            </div>
                            ${statusBadge}
                        </div>
                        <p class="text-xs text-slate-500"><strong>Indikator:</strong> ${q.indikator}</p>
                        <div class="text-slate-800 font-medium">${q.question}</div>
                        <div class="grid grid-cols-1 sm:grid-cols-2 gap-2 pt-1 text-xs">
                            <div class="p-2.5 rounded-xl bg-white border border-slate-200">
                                <span class="text-slate-500 block mb-0.5">Jawaban Siswa:</span>
                                <span class="${isCorrect ? 'text-emerald-700' : 'text-rose-700'}">${studentAnswerText}</span>
                            </div>
                            <div class="p-2.5 rounded-xl bg-white border border-slate-200">
                                <span class="text-slate-500 block mb-0.5">Kunci Jawaban:</span>
                                <span class="text-emerald-700">${keyAnswerText}</span>
                            </div>
                        </div>
                        <div class="p-3 rounded-xl bg-primary-50/60 border border-primary-100 text-xs text-slate-700 leading-relaxed">
                            <strong class="text-primary-800"><i class="fa-solid fa-lightbulb mr-1"></i>Pembahasan:</strong> ${q.explanation}
                        </div>
                    </div>
                `;
            });
        }

        // Download PDF Handler
        document.getElementById('btn-download-pdf').addEventListener('click', function() {
            const btn = this;
            const originalHTML = btn.innerHTML;
            btn.disabled = true;
            btn.innerHTML = `<i class="fa-solid fa-spinner fa-spin"></i><span>Memproses PDF...</span>`;

            const element = document.getElementById('pdf-report-area');
            const cleanName = studentData.nama.replace(/[^a-zA-Z0-9]/g, '_');
            const filename = `Hasil_TKA_Matematika_30Soal_${cleanName}_${studentData.nisn}.pdf`;

            const opt = {
                margin:       [10, 10, 10, 10],
                filename:     filename,
                image:        { type: 'jpeg', quality: 0.98 },
                html2canvas:  { scale: 2, useCORS: true, logging: false },
                jsPDF:        { unit: 'mm', format: 'a4', orientation: 'portrait' },
                pagebreak:    { mode: ['avoid-all', 'css', 'legacy'] }
            };

            html2pdf().set(opt).from(element).save().then(() => {
                btn.disabled = false;
                btn.innerHTML = originalHTML;
                showToast('Laporan PDF 30 soal berhasil diunduh!', 'success');
            }).catch(err => {
                console.error(err);
                btn.disabled = false;
                btn.innerHTML = originalHTML;
                showToast('Terjadi kendala saat membuat PDF. Coba gunakan fitur Print browser (Ctrl+P).', 'error');
            });
        });
